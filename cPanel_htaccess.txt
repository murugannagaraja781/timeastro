<IfModule mod_rewrite.c>
  RewriteEngine On

  # Define default index files
  DirectoryIndex index.html index.php

  # Route API requests to the api folder
  RewriteRule ^api/(.*)$ api/$1 [L]

  # If the requested file or directory exists, serve it directly
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  # If the requested URL matches a .html file, serve it
  RewriteCond %{DOCUMENT_ROOT}/$1.html -f
  RewriteRule ^(.*)$ $1.html [L]

  # Otherwise, fallback to index.html (useful for client-side routing if any, though Next.js static export uses individual HTML files)
  RewriteRule ^(.*)$ index.html [L]
</IfModule>
